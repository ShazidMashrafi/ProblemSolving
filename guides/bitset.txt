// ==================== STD::BITSET CHEATSHEET ====================

const int SIZE = 64;
bitset<SIZE> b;             // All bits initialized to 0
bitset<SIZE> b2(42);        // Initialized from integer
bitset<SIZE> b3("101010");  // Initialized from binary string

// Element Access & Modification
b.set(pos);           // Set bit at pos to 1
b.reset(pos);         // Reset bit at pos to 0
b.flip(pos);          // Toggle bit at pos
b[pos];               // Access bit at pos (can read/assign: b[pos] = 1)

b.set();              // Set ALL bits to 1
b.reset();            // Reset ALL bits to 0
b.flip();             // Invert ALL bits

// Queries
b.count();            // Number of set bits (popcount)
b.size();             // Total number of bits (constant: SIZE)
b.test(pos);          // Returns true if bit at pos is 1 (with bounds check)
b.any();              // Returns true if at least one bit is 1
b.none();             // Returns true if all bits are 0
b.all();              // Returns true if all bits are 1

// Conversions
string s = b.to_string();            // Convert to "000...101"
unsigned long val = b.to_ulong();    // Convert to unsigned long (throws overflow_error if > 32 bits)
unsigned long long ll_val = b.to_ullong(); // Convert to unsigned long long (up to 64 bits)

// b = ~b1;
