// ==================== BIT TRICKS & BUILTINS ====================

// --- GCC Built-in Functions ---
__builtin_popcount(n);      // Number of set bits in 32-bit int
__builtin_popcountll(n);    // Number of set bits in 64-bit int (long long)
__builtin_clz(n);           // Count leading zeros (31 - __builtin_clz(n) = highest set bit pos, 0-indexed)
__builtin_clzll(n);         // Count leading zeros for 64-bit
__builtin_ctz(n);           // Count trailing zeros (position of lowest set bit)
__builtin_ctzll(n);         // Count trailing zeros for 64-bit
__builtin_ffs(n);           // 1 + index of least significant set bit (0 if n == 0)

// --- Bit Manipulation Operations ---
// Note: ALWAYS use 1LL when shifting bits with 64-bit integers: (1LL << x)
check_bit(n, i)     : (n & (1LL << i)) != 0
set_bit(n, i)       : n | (1LL << i)
clear_bit(n, i)     : n & ~(1LL << i)
toggle_bit(n, i)    : n ^ (1LL << i)
lowest_set_bit(n)   : n & (-n)
clear_lowest_bit(n) : n & (n - 1)
n_set_bits(n)       : (1LL << n) - 1   // e.g. 3 -> 7 (0b111)

bool isPowerOfTwo(long long n) {
    return n > 0 && !(n & (n - 1));
}

// Case conversion using bitwise:
char to_lower(char c) { return c | ' '; }  // c | 32
char to_upper(char c) { return c & '_'; }  // c & (~32)

// Swap two variables without temporary:
// a ^= b; b ^= a; a ^= b;

// --- Properties of Bitwise Operations ---
// XOR Properties:
// a ^ a = 0, a ^ 0 = a
// a ^ (a & b) = (a | b) ^ b
// b ^ (a & b) = (a | b) ^ a
// (a & b) ^ (a | b) = a ^ b

// Addition & Subtraction via Bits:
// a + b = (a | b) + (a & b)
// a + b = (a ^ b) + 2 * (a & b)
// a - b = (a ^ (a & b)) - ((a | b) ^ a)
