// ================================================================
// STL COMPLEXITY & CHEATSHEET
// Rule: Any operation NOT listed below is O(1) (e.g. push_back,
// pop_back, front, back, size, empty, operator[] on vector/deque).
// ================================================================

// --- VECTOR / DEQUE / LIST ---
v.clear()                         // O(N)
v.erase(iter), v.insert(iter, x)  // O(N) - shifts subsequent elements
v.erase(start_iter, end_iter)     // O(N)

// --- PRIORITY QUEUE ---
priority_queue<int> pq;                                    // Max-heap
priority_queue<int, vector<int>, greater<int>> pq;         // Min-heap
// Custom comparator (sort by second ascending):
struct Cmp { bool operator()(const pair<int,int>& a, const pair<int,int>& b) { return a.second > b.second; } };
priority_queue<pair<int,int>, vector<pair<int,int>>, Cmp> pq;
pq.push(x), pq.pop()              // O(log N)

// --- SET / MULTISET / MAP (Red-Black Tree) ---
// insert, erase, find, count, lower_bound, upper_bound: all O(log N)
s.clear(), m.clear()              // O(N)
// TRAP 1: Never use std::lower_bound on std::set!
s.lower_bound(x);                 // O(log N) - CORRECT
std::lower_bound(s.begin(), s.end(), x); // O(N) - TLE! (Iterators cannot jump)
// TRAP 2: multiset::erase(val) deletes ALL copies!
ms.erase(val);                    // O(log N + count) - erases ALL instances of val
ms.erase(ms.find(val));           // O(log N) - erases ONLY ONE instance

// --- UNORDERED SET / MAP (Hash Table) ---
// insert, erase, find, count: O(1) average, O(N) worst-case under collisions

// --- STRING OPERATIONS ---
s += c, s.push_back(c)            // O(1) amortized
s = s + c                         // O(N) - Copies entire string! TLE in loops!
s.substr(pos, len)                // O(len) - Copies characters
s.find(pat), s.rfind(pat)         // O(N * M)
s.replace(), s.insert(), s.erase()// O(N)
stoi(s), to_string(x)             // O(len)

// --- ESSENTIAL ALGORITHMS ---
nth_element(a.begin(), a.begin() + k, a.end()); // O(N) average: puts k-th smallest at index k
lower_bound(a.begin(), a.end(), x);   // O(log N) on sorted random-access
upper_bound(a.begin(), a.end(), x);   // O(log N) on sorted random-access
unique(a.begin(), a.end());           // O(N) - requires sorted array
// Remove duplicates idiom:
a.erase(unique(a.begin(), a.end()), a.end());   // O(N)
accumulate(a.begin(), a.end(), 0LL);  // O(N) - use 0LL to prevent 32-bit overflow
