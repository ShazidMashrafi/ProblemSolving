================================================================
MULTI-TESTCASE GRAPHS GUIDE
================================================================

1. RESET ONLY UP TO N (AVOID O(T * MAXN) TLE)
   for (int i = 1; i <= n; ++i) {
       g[i].clear();
       dist[i] = -1;
       vis[i] = 0;
   }
   // Never: memset(arr, 0, sizeof(arr))

2. READ ALL INPUT BEFORE EARLY EXIT
   // BAD:  if (bad) return; // unread input corrupts next test!
   // GOOD: read all m edges first, then return/print.
   for (int i = 0; i < m; ++i) cin >> u >> v;
   if (bad) { cout << -1 << '\n'; return; }

3. GLOBAL STATIC ARRAYS (FASTER THAN LOCAL VECTORS)
   // Local vector<int> g[n+1] inside solve() causes heap churn.
   // Use global vector<int> g[N] and g[i].clear() up to n.

4. 2D GRID RESET (O(N * M))
   for (int i = 0; i < n; ++i) fill(dist[i], dist[i] + m, -1);

5. PASS BY REFERENCE (AVOID O(V + E) COPY)
   void dfs(int u, const vector<vector<int>>& adj);
